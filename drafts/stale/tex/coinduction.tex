\[
\begin{alignedat}{2}
    F(\emptyset)        &= \{A\}         \\
    F(\{A\})            &= \{A\}         \\
    F(\{B\})            &= \{A\}         \\
    F(\{C\})            &= \{A\}         \\
    F(\{A, B\})         &= \{A, B\}      \\
    F(\{A, C\})         &= \{A, B\}      \\
    F(\{B, C\})         &= \{A\}         \\
    F(\{A, B, C\})      &= \{A, B, C\}
\end{alignedat}
\]
